# 1. Обробка вебхуків (Python / FastAPI)
from fastapi import FastAPI, Request, Header
import hmac
import hashlib

app = FastAPI()
TWITCH_SECRET = "your_secret_key_here"

@app.post("/eventsub/callback")
async def eventsub_handler(request: Request, 
                           twitch_eventsub_message_signature: str = Header(None)):
    
    body = await request.body()
    expected_signature = "sha256=" + hmac.new(
        TWITCH_SECRET.encode(), body, hashlib.sha256
    ).hexdigest()

    if not hmac.compare_digest(expected_signature, twitch_eventsub_message_signature):
        return {"status": "403 Forbidden"}

    data = await request.json()
    
    if data.get("subscription", {}).get("type") == "stream.online":
        broadcaster = data["event"]["broadcaster_user_name"]
        print(f"ПОДІЯ: Стрімер {broadcaster} щойно запустив трансляцію!")
        
    return {"status": "success"}

    const message = data.toString();

        const username = message.split('!')[0].substring(1);
        const text = message.split('PRIVMSG')[1].split(':').slice(1).join(':').trim();
        console.log(`[Чат] ${username}: ${text}`);
    }
    
    if (message.startsWith('PING')) {
        ws.send('PONG :tmi.twitch.tv');
    }
});

import redis
import json

cache = redis.Redis(host='localhost', port=6379, db=0)

def get_streamer_status(user_id):
    cache_key = f"streamer_status:{user_id}"
    
    cached_data = cache.get(cache_key)
    
    if cached_data:
        print("Отримано з Redis (Швидко)")
        return json.loads(cached_data)
        
    print("Дані відсутні в кеші. Запит до бази даних...")
    db_data = fetch_from_postgres(user_id)
    
    if db_data:
        cache.setex(cache_key, 60, json.dumps(db_data))
